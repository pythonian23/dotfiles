bindsym {
        $MOD+M        exec printf "%s\n" "🧹 Clear" "⏯️ Play/Pause" "⏮️ Previous" "⏭️ Next" "🔇 Mute" "🔈 Low Volume" "🔉 Medium Volume" "🔊 High Volume" "📢 Max Volume" "🎲 Random" | tofi --prompt-text "🎸 " | awk '{ print $1 }' | sed -e "s/🧹/clear/ ; s/⏯️/toggle/ ; s/⏮️/prev/ ; s/⏭️/next/ ; s/🔇/volume 0/ ; s/🔈/volume 25/ ; s/🔉/volume 50/ ; s/🔊/volume 75/ ; s/📢/volume 100/ ; s/🎲/random/" | xargs mpc
        $MOD_S+M  exec mpc list Album | tofi --prompt-text '💿 ' | xargs -I % mpc searchadd Album "%" && mpc play
